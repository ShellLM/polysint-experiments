Here is the `summary.py` script. This script analyzes the `analyzer.log` file generated by the system, aggregates statistics (errors, warnings, suppressed alerts), and broadcasts a "Daily Intelligence Digest" to your configured Discord/Telegram channels.

It is designed to be run once a day via a cron job or scheduled task.

### FILE: summary.py ###

```python
import os
from datetime import datetime, timedelta
from logger import get_logger
from notifier import Notifier

log = get_logger("Summary")

# Path to the log file created by logger.py
LOG_FILE = "analyzer.log"

def generate_digest():
    """
    Parses the log file for entries from the last 24 hours.
    Aggregates system health metrics and noise filtration stats.
    """
    if not os.path.exists(LOG_FILE):
        print("Log file not found. Nothing to summarize.")
        return

    # Time window: last 24 hours
    cutoff = datetime.now() - timedelta(hours=24)
    
    stats = {
        "errors": 0,
        "warnings": 0,
        "suppressed_alerts": 0,
        "api_failures": 0,
        "harvest_glitches": 0,
        "start_time": None,
        "end_time": None
    }

    # Read log file
    try:
        with open(LOG_FILE, 'r', encoding='utf-8', errors='ignore') as f:
            lines = f.readlines()
    except Exception as e:
        print(f"Failed to read log file: {e}")
        return

    for line in lines:
        try:
            # Log format: 2023-10-27 10:00:00,123 - Module - LEVEL - Message
            parts = line.split(" - ")
            if len(parts) < 4:
                continue

            # Parse timestamp (strip milliseconds for standard parsing)
            timestamp_str = parts[0].split(',')[0]
            ts = datetime.strptime(timestamp_str, "%Y-%m-%d %H:%M:%S")

            if ts < cutoff:
                continue

            # Track time range of the data we actually processed
            if stats["start_time"] is None:
                stats["start_time"] = ts
            stats["end_time"] = ts

            level = parts[2].strip()
            module = parts[1].strip()
            message = ' - '.join(parts[3:]).strip()

            if level == "ERROR":
                stats["errors"] += 1
                if "API" in module or "Profile" in message or "Analysis" in message:
                    stats["api_failures"] += 1

            elif level == "WARNING":
                stats["warnings"] += 1
                
                # Specific intelligence extraction
                if "Suppressed alert" in message:
                    stats["suppressed_alerts"] += 1
                elif "Harvester" in module or "Network glitch" in message:
                    stats["harvest_glitches"] += 1

        except (ValueError, IndexError):
            # Skip malformed lines or stack traces
            continue

    return stats

def broadcast_digest(stats):
    """
    Formats the stats into an intelligence digest and broadcasts it.
    """
    if not stats or stats["start_time"] is None:
        print("No log entries found in the last 24 hours.")
        return

    # Calculate health score (simple heuristic)
    # Deduct points for errors and glitches
    health_score = 100 - (stats["errors"] * 5) - (stats["harvest_glitches"] * 2)
    health_score = max(0, health_score)

    if health_score >= 90:
        health_emoji = "🟢"
        status = "Nominal"
    elif health_score >= 70:
        health_emoji = "🟡"
        status = "Degraded"
    else:
        health_emoji = "🔴"
        status = "Critical"

    msg = (
        f"**{health_emoji} System Health: {status} ({health_score}/100)**\n"
        f"━━━━━━━━━━━━━━━━━━━━━━━━━━━━\n"
        f"⏰ **Window:** {stats['start_time'].strftime('%b %d, %H:%M')} - {stats['end_time'].strftime('%H:%M')} UTC\n\n"
        f"**Operational Metrics**\n"
        f"• **Errors:** {stats['errors']}\n"
        f"• **Warnings:** {stats['warnings']}\n"
        f"• **API Failures:** {stats['api_failures']}\n"
        f"• **Harvest Glitches:** {stats['harvest_glitches']}\n\n"
        f"**Intelligence Filtration**\n"
        f"• **Suppressed Alerts (Noise):** {stats['suppressed_alerts']}\n"
        f"_Markets near resolution or low volume ignored._\n\n"
        f"━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
    )

    print(msg)
    notifier = Notifier()
    notifier.broadcast(msg, title="📊 Daily Intelligence Digest")

if __name__ == "__main__":
    print("Generating daily intelligence digest...")
    stats = generate_digest()
    if stats:
        broadcast_digest(stats)
```

### Integration Instructions

To automate this report, add it to your system's crontab (Linux/macOS) or Task Scheduler (Windows).

**Example Crontab (Runs daily at 00:00 UTC):**
```bash
0 0 * * * cd /path/to/PolySINT && /usr/bin/python3 summary.py >> logs/summary.log 2>&1
```
